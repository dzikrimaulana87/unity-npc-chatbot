    private IEnumerator Start()
    {
        // Tetapkan hanya state Happy
        npcContext.MakeHappy();
        Debug.Log("====== TESTING STATE: Happy ======");

        int correct = 0;
        foreach (var (input, expectedTag) in testSet)
        {
            string predictedTag = npcContext.TestInput(input); // diasumsikan metode ini mengembalikan `matchedCategory.tag`
            bool match = predictedTag == expectedTag;

            Debug.Log($"Input: \"{input}\" → Predicted: \"{predictedTag}\" | Expected: \"{expectedTag}\" → {(match ? "✅" : "❌")}");

            if (match) correct++;

            yield return new WaitForSeconds(0.05f); // percepat interval log
        }

        float accuracy = (float)correct / testSet.Count * 100f;
        Debug.Log($"🎯 Akurasi: {accuracy:F2}% dari {testSet.Count} input");
        Debug.Log("=== Pengujian selesai ===");
    }
